function drawGaugeConseilMunicipal(elementId, value) {
const data = [
    {
      type: "indicator",
      mode: "gauge+number",
      value: value,
      number: {
        suffix: " sièges",
        font: { size: 20 },
      },
      gauge: {
        axis: {
          range: [0, 29],
          tickvals: [0, 15, 22, 27, 29],
          ticktext: ["0", "15", "22", "27", "29"],
          tickwidth: 1,
          tickcolor: "#888",
        },
        bar: {
          color: "#1a1a1a",
          thickness: 0.03,
        },
        steps: [
          { range: [0,  15], color: "#185FA5" }, // Majorité absolue
          { range: [15, 22], color: "#1D9E75" }, // 51%
          { range: [22, 27], color: "#EF9F27" }, // 35%
          { range: [27, 29], color: "#E24B4A" }, // 14%
        ],
        threshold: {
          line: { color: "#1a1a1a", width: 3 },
          thickness: 0.75,
          value: value,
        },
      },
    },
  ];
 
  const segments = [
    { label: "Majorité absolue (0–14)", color: "#185FA5" },
    { label: "51 % des sièges (15–21)",  color: "#1D9E75" },
    { label: "35 % des sièges (22–26)",  color: "#EF9F27" },
    { label: "14 % des sièges (27–29)",  color: "#E24B4A" },
  ];
 
  const layout = {
    width: 700,
    height: 620,
    annotations: [],
    showlegend: true,
    legend: {
      orientation: "h",
      x: 0.5,
      xanchor: "center",
      y: -0.05,
      font: { size: 11 },
    },
    paper_bgcolor: "white",
    plot_bgcolor: "white",
        xaxis: { visible: false, showgrid: false, zeroline: false },
        yaxis: { visible: false, showgrid: false, zeroline: false },
    font: { family: "sans-serif" },
  };
 
  // Plotly indicator ne supporte pas la légende native,
  // on ajoute des traces invisibles pour forcer l'affichage.
  segments.forEach(seg => {
    data.push({
      type: "scatter",
      x: [null],
      y: [null],
      mode: "markers",
      marker: { color: seg.color, size: 10, symbol: "square" },
      name: seg.label,
      showlegend: true,
    });
  });

  const config = {
            responsive: true,
            displayModeBar: false // Optionnel : masque la barre d'outils Plotly pour un rendu plus propre sur vos slides
        };

    // 4. On retourne l'objet JSON attendu par Plotly / SliDesk
    return {
        data,
        layout,
        config
    };
}
